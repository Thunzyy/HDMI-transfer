SQL

CREATE TABLE "artists" (
	"id"	INTEGER,
	"name"	TEXT,
	"distributor"	TEXT,
	"total listens"	INTEGER,
	"listen today"	INTEGER,
	"spotify link"	TEXT,
	"mail"	TEXT,
	"password"	TEXT,
	"first name"	TEXT,
	"last name"	TEXT,
	"creation date"	DATE,
	"account created"	BOOLEAN,
	"password_sfa"	TEXT,
	PRIMARY KEY("id" AUTOINCREMENT)
);

CREATE TABLE "albums" (
	"id"	INTEGER,
	"artist_id"	INTEGER,
	"title"	TEXT,
	"link"	TEXT,
	"songs_on_album"	INTEGER,
	PRIMARY KEY("id" AUTOINCREMENT),
	FOREIGN KEY("artist_id") REFERENCES "artists"("id") ON DELETE CASCADE
);

CREATE TABLE "songs" (
	"id"	INTEGER,
	"artist_id"	INTEGER,
	"album_id"	INTEGER,
	"title"	TEXT,
	"link"	TEXT,
	PRIMARY KEY("id" AUTOINCREMENT),
	FOREIGN KEY("album_id") REFERENCES "albums"("id") ON DELETE CASCADE,
	FOREIGN KEY("artist_id") REFERENCES "artists"("id") ON DELETE CASCADE
);

CREATE TABLE "playlists" (
	"id"	INTEGER,
	"spotify_account_id"	INTEGER,
	"title"	TEXT,
	"link"	TEXT,
	"songs_on_playlist"	INTEGER,
	PRIMARY KEY("id" AUTOINCREMENT),
	FOREIGN KEY("spotify_account_id") REFERENCES "spotify_accounts"("ID") ON DELETE CASCADE
);

CREATE TABLE "devices" (
	"id"	INTEGER,
	"deviceid"	TEXT NOT NULL,
	"spotify_account_id"	INTEGER,
	"adb_connected"	INTEGER,
	"active_listening"	INTEGER,
	PRIMARY KEY("id" AUTOINCREMENT)
);

CREATE TABLE "spotify_accounts" (
	"ID"	INTEGER,
	"Email"	TEXT NOT NULL,
	"Pseudo"	TEXT NOT NULL,
	"Password"	TEXT NOT NULL,
	"Country_code"	TEXT NOT NULL,
	"Type_of_account"	TEXT NOT NULL,
	"Deviceid_linked"	INTEGER,
	"Artist_listening_now"	INTEGER,
	"Settings_edited"	TEXT NOT NULL,
	"liked_song"	INTEGER NOT NULL,
	"Listen_Time"	INTEGER NOT NULL,
	"Listen_count_today"	INTEGER NOT NULL,
	"Total_Listen_Count"	INTEGER NOT NULL,
	"Gender"	TEXT NOT NULL,
	"Birthday_date"	TEXT NOT NULL,
	"Creation_date"	TEXT NOT NULL,
	"Expiration_date"	TEXT NOT NULL,
	"Artist_time_listen"	INTEGER NOT NULL,
	"Other_artist_time_listen"	INTEGER NOT NULL,
	"Disconected_count"	INTEGER NOT NULL,
	"Private_playlist"	TEXT,
	"songs_on_private_playlist"	INTEGER,
	PRIMARY KEY("ID"),
	FOREIGN KEY("Artist_listening_now") REFERENCES "artists"("id") ON DELETE CASCADE
);
