Create ThreadedCapture that wraps CaptureSource in a background thread with a ring buffer, eliminating blocking cap.read() stalls in the main decode thread. Add FPSReporter for real-time FPS measurement and reporting.

Purpose: The blocking cap.read() call stalls the main thread while the USB/HDMI capture card delivers the next frame. A background capture thread with a deque ring buffer decouples capture from processing, allowing the decoder to work on the latest frame without waiting. FPS reporting gives visibility into actual capture performance.
Output: threaded.py with ThreadedCapture and FPSReporter, comprehensive unit tests.

    ```python
    class FPSReporter:
        """Measures and reports actual FPS over a sliding window.

        Call tick() once per frame. It returns the measured FPS when the
        reporting interval elapses, otherwise None.

        Parameters
        ----------
        report_interval_s:
            Seconds between FPS reports (default 1.0).
        """

        def __init__(self, report_interval_s: float = 1.0) -> None:
            self._interval_ns = int(report_interval_s * 1e9)
            self._window_frames = 0
            self._window_start_ns = time.perf_counter_ns()
            self._total_frames = 0

        def tick(self) -> float | None:
            """Record one frame. Returns FPS when interval elapses, else None."""
            self._window_frames += 1
            self._total_frames += 1
            now = time.perf_counter_ns()
            elapsed = now - self._window_start_ns
            if elapsed >= self._interval_ns:
                fps = self._window_frames / (elapsed / 1e9)
                self._window_frames = 0
                self._window_start_ns = now
                return fps
            return None

        @property
        def total_frames(self) -> int:
            return self._total_frames
    ```

    2. ThreadedCapture class:
    ```python
    class ThreadedCapture:
        """Background-threaded video capture with ring buffer.

        Wraps a CaptureSource (or any object with a .read() method returning
        (bool, frame)) in a daemon thread that continuously captures frames
        into a bounded deque.

        Parameters
        ----------
        source:
            Object with .read() -> (bool, ndarray) method (typically CaptureSource).
        buffer_size:
            Maximum frames in the ring buffer (default 16). Oldest frames
            are auto-discarded when full. 16 frames at 240fps = ~67ms buffer.
        """

        def __init__(self, source, buffer_size: int = 16) -> None:
            self._source = source
            self._buffer: deque = deque(maxlen=buffer_size)
            self._stopped = False
            self._thread = threading.Thread(target=self._capture_loop, daemon=True)
            self._fps_reporter = FPSReporter()

        def start(self):
            """Start the capture thread. Returns self for chaining."""
            self._thread.start()
            return self

        def _capture_loop(self) -> None:
            """Background loop: continuously read frames into buffer."""
            while not self._stopped:
                ret, frame = self._source.read()
                if ret:
                    self._buffer.append(frame)
                    self._fps_reporter.tick()
                # If read fails, loop retries immediately (no sleep --
                # cap.read() itself blocks until next frame)

        def read(self) -> tuple[bool, object]:
            """Non-blocking read from buffer.

            Returns
            -------
            tuple[bool, numpy.ndarray | None]
                (True, frame) if a frame is available, (False, None) if
                the buffer is empty.
            """
            try:
                return True, self._buffer.popleft()
            except IndexError:
                return False, None

        @property
        def actual_fps(self) -> float:
            """Measured capture FPS from the background thread."""
            total = self._fps_reporter.total_frames
            if total == 0:
                return 0.0
            # Calculate from reporter's internal state
            now = time.perf_counter_ns()
            start = self._fps_reporter._window_start_ns
            elapsed = (now - start) / 1e9
            if elapsed <= 0:
                return 0.0
            return self._fps_reporter._window_frames / elapsed if elapsed < 2.0 else 0.0

        @property
        def capture_fps_reporter(self) -> FPSReporter:
            """Direct access to the FPSReporter for the capture thread."""
            return self._fps_reporter

        @property
        def buffer_size(self) -> int:
            """Current number of frames in the buffer."""
            return len(self._buffer)

        def stop(self) -> None:
            """Signal the capture thread to stop and wait for it."""
            self._stopped = True
            if self._thread.is_alive():
                self._thread.join(timeout=2.0)

        def __enter__(self):
            self.start()
            return self

        def __exit__(self, exc_type, exc_val, exc_tb):
            self.stop()
    ```

    Imports needed: `from __future__ import annotations`, `import threading`, `import time`, `from collections import deque`.

    Do NOT import CaptureSource -- ThreadedCapture uses duck typing (any object with .read()). This avoids circular imports and allows wrapping mock sources in tests.
