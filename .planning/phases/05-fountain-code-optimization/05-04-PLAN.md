    ```python
    def measure_overhead(K, payload_size, runs=20, max_multiplier=5):
        """Measure average overhead for given K across multiple runs.

        Returns (avg_overhead_ratio, min_overhead, max_overhead, all_results).
        overhead_ratio = droplets_needed / K (1.0 = perfect, 1.1 = 10% overhead).
        """
        results = []
        for run in range(runs):
            data = os.urandom(K * payload_size)
            chunks = [prepare chunk i for i in range(K)]
            decoder = FountainDecoder(K, payload_size)

            seed = run * 10000 + 1  # deterministic but varied
            droplets = 0
            while not decoder.is_complete() and droplets < K * max_multiplier:
                payload = build_droplet(seed, K, chunks)
                decoder.add_droplet(seed, payload)
                droplets += 1
                seed += 1

            assert decoder.is_complete()
            results.append(droplets / K)

        return (mean(results), min(results), max(results), results)
    ```
